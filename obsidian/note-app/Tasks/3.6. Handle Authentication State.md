### 1. Create Authentication Store

First, let's create a dedicated store for managing authentication state:

```javascript:src/stores/auth.js
import { reactive, readonly } from 'vue'
import api from '@/services/api'

const state = reactive({
  token: localStorage.getItem('token') || null,
  user: null,
  loading: false,
  error: null
})

const actions = {
  setToken(token) {
    state.token = token
    if (token) {
      localStorage.setItem('token', token)
    } else {
      localStorage.removeItem('token')
    }
  },

  setUser(user) {
    state.user = user
  },

  async login(credentials) {
    state.loading = true
    state.error = null
    try {
      const response = await api.post('/auth/token', credentials)
      const { access_token, user } = response.data
      this.setToken(access_token)
      this.setUser(user)
      return true
    } catch (error) {
      state.error = error.response?.data?.detail || 'Login failed'
      return false
    } finally {
      state.loading = false
    }
  },

  async logout() {
    this.setToken(null)
    this.setUser(null)
  },

  async checkAuth() {
    if (!state.token) return false
    try {
      const response = await api.get('/auth/me')
      this.setUser(response.data)
      return true
    } catch (error) {
      this.logout()
      return false
    }
  }
}

export default {
  state: readonly(state),
  ...actions
}
```

### 2. Configure Axios Interceptors

Update your API service to include authentication interceptors:

```javascript:src/services/api.js
import axios from 'axios'
import auth from '@/stores/auth'
import router from '@/router'

const api = axios.create({
  baseURL: 'http://localhost:8000',
  headers: {
    'Content-Type': 'application/json'
  }
})

// Request interceptor
api.interceptors.request.use(
  (config) => {
    const token = auth.state.token
    if (token) {
      config.headers.Authorization = `Bearer ${token}`
    }
    return config
  },
  (error) => {
    return Promise.reject(error)
  }
)

// Response interceptor
api.interceptors.response.use(
  (response) => response,
  (error) => {
    if (error.response?.status === 401) {
      auth.logout()
      router.push('/login')
    }
    return Promise.reject(error)
  }
)

export default api
```

### 3. Update Login Component

```vue:src/views/Login.vue
<script>
import { ref, reactive } from 'vue'
import { useRouter } from 'vue-router'
import auth from '@/stores/auth'

export default {
  name: 'Login',
  setup() {
    const router = useRouter()
    const loading = ref(false)
    const form = reactive({
      email: '',
      password: ''
    })
    const errors = reactive({
      email: '',
      password: '',
      general: ''
    })

    const handleSubmit = async () => {
      if (!validateForm()) return

      loading.value = true
      try {
        const success = await auth.login({
          username: form.email,
          password: form.password
        })
        
        if (success) {
          router.push('/notes')
        }
      } catch (error) {
        errors.general = 'Login failed. Please try again.'
      } finally {
        loading.value = false
      }
    }

    // ... rest of the component code
  }
}
</script>
```

### 4. Update Navigation Guard

Modify your router configuration to use the auth store:

```javascript:src/router/index.js
import auth from '@/stores/auth'

// Add before the router creation
router.beforeEach(async (to, from, next) => {
  if (to.meta.requiresAuth) {
    const isAuthenticated = await auth.checkAuth()
    if (!isAuthenticated) {
      next('/login')
      return
    }
  }
  next()
})
```

### 5. Update NavBar Component

Update the navigation bar to use the auth store for logout:

```vue:src/components/NavBar.vue
<script>
import { computed } from 'vue'
import { useRouter } from 'vue-router'
import auth from '@/stores/auth'

export default {
  name: 'NavBar',
  setup() {
    const router = useRouter()
    const isAuthenticated = computed(() => !!auth.state.token)

    const handleLogout = async () => {
      await auth.logout()
      router.push('/login')
    }

    return {
      isAuthenticated,
      handleLogout
    }
  }
}
</script>
```
